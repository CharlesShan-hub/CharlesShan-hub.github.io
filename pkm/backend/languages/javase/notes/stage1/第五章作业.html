编程实现如下功能 某人有 100000 元，每次经过一次路口，需要缴费： 1. 现金>5w时，每次交5% 2. 现金≤ 5w 时，每次交 1000 计算这个人可以经过多少次路口，用 while break

public class Ex1{
    public static void main(String[] args){
        double money = 100000.0;
        int count = 0;
        while(money>0){
            if(money > 50000)
                money *= 0.95;
            else if(money < 1000)
                break;
            else
                money -= 1000;
            count++;
        }
        System.out.println(count); // 62
    }
}

        for(int i=1, count=1; i<=100; i++, count++){
            if(i%5==0)
                continue;
            System.out.print(i+"\t");
            if(count==5){
                count = 0;

    }

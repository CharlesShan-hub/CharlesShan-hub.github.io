class Season{  
  private String name;  
  private String decs;  
  public Season(String name, String decs){  
    this.name = name;  
    this.decs = decs;  
  }  
  public String getDecs() {  
    return decs;  
  }  
  public String getName() {  
    return name;  
  }  
  public void setName(String name) {  
    this.name = name;  
  }  
  public void setDecs(String decs) {  
    this.decs = decs;  
  }
}

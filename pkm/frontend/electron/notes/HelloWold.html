使用预加载脚本

  1. 创建preload.js
const { contextBridge } = require('electron')

contextBridge.exposeInMainWorld('versions', {
  node: () => process.versions.node,
  chrome: () => process.versions.chrome,
  electron: () => process.versions.electron,
  // 能暴露的不仅仅是函数，我们还可以暴露变量
})
  1. main.js中添加const path = require('path')与webPreferences: {preload: path.join(__dirname, 'preload.js'),}
// main.js
const { app, BrowserWindow } = require('electron')
const path = require('path')

const createWindow = () => {
  const win = new BrowserWindow({
    width: 800,
    height: 600,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
    },

  win.loadFile('index.html')
}

  createWindow()
})

在进程之间通信

  1. 修改preload.js
const { contextBridge, ipcRenderer } = require('electron')

contextBridge.exposeInMainWorld('versions', {
  node: () => process.versions.node,
  chrome: () => process.versions.chrome,
  electron: () => process.versions.electron,
  // 能暴露的不仅仅是函数，我们还可以暴露变量
})

contextBridge.exposeInMainWorld('connection', {
  ping: () => ipcRenderer.invoke('ping')
})
  1. 修改renderer.js
const func = async () => {
  const response = await window.connection.ping()
  alert(response) // 弹窗 'pong'
}

func()
  1. 在main.js中引入ipcmain模块const { app, BrowserWindow, ipcMain } = require('electron')，并加入函数ipcMain.handle('ping', () => 'pong')
const { app, BrowserWindow, ipcMain } = require('electron')
const path = require('path')

const createWindow = () => {
  const win = new BrowserWindow({
    width: 800,
    height: 300,
    webPreferences: {
      preload: path.join(__dirname, 'preload.js'),
    },
  })
  ipcMain.handle('ping', () => 'pong')
  win.loadFile('index.html')
}
app.whenReady().then(createWindow)
  1. 运行npm start，运行结果
